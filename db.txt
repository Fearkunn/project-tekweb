CREATE TABLE users (
id_user INT AUTO_INCREMENT PRIMARY KEY,
username VARCHAR(255) NOT NULL,
user_password VARCHAR(255) NOT NULL,
user_email VARCHAR(255) NOT NULL,
user_profile VARCHAR(255), 
role_user VARCHAR(50)
);

CREATE TABLE publisher (
id_publisher INT AUTO_INCREMENT PRIMARY KEY,
publisher_name VARCHAR(255) NOT NULL,
publisher_password VARCHAR(255) NOT NULL,
publisher_logo VARCHAR(255),
publisher_email VARCHAR(255)
);

CREATE TABLE games (
id_game INT AUTO_INCREMENT PRIMARY KEY,
game_name VARCHAR(255) NOT NULL,
game_desc TEXT,
is_admit BOOLEAN NOT NULL,
release_date DATE,
like_count INT DEFAULT 0,
games_image VARCHAR(255),
id_publisher INT,
FOREIGN KEY (id_publisher) REFERENCES publisher(id_publisher) ON DELETE CASCADE
);

CREATE TABLE library (
id_library INT AUTO_INCREMENT PRIMARY KEY,
id_user INT,
id_game INT,
is_like INT,
FOREIGN KEY (id_user) REFERENCES users(id_user) ON DELETE CASCADE,
FOREIGN KEY (id_game) REFERENCES games(id_game) ON DELETE CASCADE
);

CREATE TABLE review (
id_review INT AUTO_INCREMENT PRIMARY KEY,
review_content TEXT,
id_user INT,
id_game INT,
FOREIGN KEY (id_user) REFERENCES users(id_user) ON DELETE CASCADE,
FOREIGN KEY (id_game) REFERENCES games(id_game) ON DELETE CASCADE
);

CREATE TABLE genre (
id_genre INT AUTO_INCREMENT PRIMARY KEY,
genre_name VARCHAR(255) NOT NULL
);

CREATE TABLE detail_genre (
id_detail INT AUTO_INCREMENT PRIMARY KEY,
id_game INT,
id_genre INT,
FOREIGN KEY (id_game) REFERENCES games(id_game) ON DELETE CASCADE,
FOREIGN KEY (id_genre) REFERENCES genre(id_genre) ON DELETE CASCADE
);
